@OptIn(ExperimentalComposeUiApi::class)
@Composable
private fun StageContent(
    item: MediaItem,
    active: Boolean,
    state: PhotoViewerState,
    imageFocus: FocusRequester,
    onOpenVideo: () -> Unit,
    onToggleChrome: () -> Unit,
    onPauseSlideshow: () -> Unit,
    onNext: () -> Unit,
    onPrev: () -> Unit,
    onEnterToolbar: () -> Unit,
    onFocusChanged: (Boolean) -> Unit,
    inToolbar: () -> Boolean,
) {
    val zoomed = active && state.zoomed
    val stage = Modifier
        .fillMaxSize()
        .then(if (active) Modifier.focusRequester(imageFocus) else Modifier)
        .onFocusChanged { onFocusChanged(it.isFocused) }
        .focusable()
        .onPreviewKeyEvent { event ->
            if (!active || event.type != KeyEventType.KeyDown) return@onPreviewKeyEvent false
            when (event.key) {
                Key.DirectionLeft, Key.DirectionRight, Key.DirectionUp, Key.DirectionDown,
                Key.DirectionCenter, Key.Enter, Key.NumPadEnter, Key.Tab, Key.Menu,
                -> Unit
                else -> return@onPreviewKeyEvent false
            }
            // 鐒︾偣鍦ㄦ寜閽鏃讹紝鏂瑰悜閿氦缁?Compose 鐒︾偣绯荤粺锛堣鎸夐挳涔嬮棿鑳借蛋锛?            if (inToolbar() && event.key != Key.DirectionCenter &&
                event.key != Key.Enter && event.key != Key.NumPadEnter &&
                event.key != Key.Tab && event.key != Key.Menu
            ) {
                return@onPreviewKeyEvent false
            }

            // 浠绘剰鏂瑰悜閿細鏆傚仠骞荤伅鐗囷紙鍚屼竴鎸夐敭缁х画鎵ц鍒囨崲鍔ㄤ綔锛?            if (event.key == Key.DirectionLeft || event.key == Key.DirectionRight ||
                event.key == Key.DirectionUp || event.key == Key.DirectionDown
            ) {
                onPauseSlideshow()
            }

            if (item.isVideo()) {
                // 瑙嗛锛氬彧鏈?ENTER 鏈夋剰涔?                if (event.key == Key.DirectionCenter || event.key == Key.Enter || event.key == Key.NumPadEnter) {
                    onOpenVideo()
                    return@onPreviewKeyEvent true
                }
                return@onPreviewKeyEvent false
            }

            when (event.key) {
                Key.DirectionLeft -> {
                    if (zoomed) state.panBy(-PanStep, 0f) else onPrev()
                    true
                }

                Key.DirectionRight -> {
                    if (zoomed) state.panBy(PanStep, 0f) else onNext()
                    true
                }

                Key.DirectionUp -> {
                    if (zoomed) state.panBy(0f, -PanStep) else true
                    true
                }

                Key.DirectionDown -> {
                    if (zoomed) {
                        state.panBy(0f, PanStep)
                        true
                    } else {
                        onEnterToolbar()
                        true
                    }
                }

                Key.DirectionCenter, Key.Enter, Key.NumPadEnter -> {
                    onToggleChrome()
                    true
                }

                Key.Tab, Key.Menu -> {
                    onEnterToolbar()
                    true
                }

                else -> false
            }
        }

    Box(stage, contentAlignment = Alignment.Center) {
        if (item.isVideo()) {
            VideoStage(item = item)
        } else {
            PhotoStage(item = item, zoomed = zoomed, panX = state.panX, panY = state.panY)
        }
    }
}

 * 缁欐灏哄锛堝師鏉ョ殑 960x540dp锛夊張浼氬鑷村浘鐗囧懆鍥寸暀鍑虹伆鑹茶垶鍙板簳銆傛墍浠ヨ繖閲岋細
 * - 鍥剧墖鏈韩 `aspectRatio(16/9)` + `fillMaxSize`锛氬緱鍒般€屽～婊″彲鐢ㄥ尯鍩熺殑鏈€澶?16:9 鐭╁舰銆嶏紝
 *   閰嶅悎 ContentScale.Fit锛?080p 绱犳潗姝ｅソ閾烘弧锛岀珫鍥?瓒呭鍥句篃鍦ㄦ鍐呯瓑姣斿眳涓紱
 * - 澶栭潰鍙敾涓€涓€屽惈鍥剧墖鍖哄煙鐨勫渾瑙掓弿杈广€嶄綔涓鸿垶鍙版劅锛屼笉閾哄簳鑹诧紝閬垮厤鍑虹幇鐏拌壊鐣欑櫧鏉°€? */
@Composable
private fun PhotoStage(item: MediaItem, zoomed: Boolean, panX: Float, panY: Float) {
    Box(
        modifier = Modifier
            .fillMaxSize()
            .padding(horizontal = TvDimen.SpaceXl, vertical = TvDimen.SpaceM),
        contentAlignment = Alignment.Center,
    ) {
        MediaImage(
            url = item.url,
            contentDescription = item.title,
            modifier = Modifier
                .aspectRatio(16f / 9f)
                .fillMaxSize()
                .then(if (zoomed) Modifier else Modifier.clip(RoundedCornerShape(TvDimen.RadiusL))),
            contentScale = ContentScale.Fit,
            targetWidth = 1600,
        )
        if (zoomed) {
            ZoomHint(
                panX = panX,
                panY = panY,
                modifier = Modifier
                    .align(Alignment.TopCenter)
                    .padding(top = TvDimen.SpaceS),
            )
        }
    }
}

@Composable
private fun SlideshowProgressBar() {
    val progress = remember { Animatable(0f) }
    LaunchedEffect(Unit) {
        while (true) {
            progress.snapTo(0f)
            progress.animateTo(1f, tween(SlideshowIntervalMs.toInt(), easing = LinearEasing))
        }
    }
    Spacer(Modifier.width(TvDimen.SpaceS))
    Box(
        Modifier
            .width(96.dp)
            .height(4.dp)
            .background(TvColor.Divider, RoundedCornerShape(2.dp)),
    ) {
        Box(
            Modifier
                .fillMaxWidth(progress.value)
                .height(4.dp)
                .background(TvColor.Accent, RoundedCornerShape(2.dp)),
        )
    }
}
